void _autoBatchRefresh() {
  if (_isNotificationScheduled) return;  // ← drop-path
  _isNotificationScheduled = true;
  scheduleMicrotask(() {                 // ← async: fires after the sync block
    refresh();                           //    Obx updaters (1×)
    notifyStream();                      //    ever/once/debounce (1×)
    _isNotificationScheduled = false;
  });
}
