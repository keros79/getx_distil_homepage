// ==========================================
// 1. RxSList (Status-Aware List)
// ==========================================
final items = <String>[].ops;        // starts as idle
final seeded = ['a', 'b'].ops;       // starts as loaded (v2.0.0+: seeded data = loaded)

// UI Binding using items.on(...) inside Obx
Obx(() => items.on(
  idle: () => const Text('Idle...'),
  loading: () => const CircularProgressIndicator(),
  loaded: (data) => ListView(children: data.map((e) => Text(e)).toList()),
  empty: () => const Text('No items found.'),
  error: (err) => Text('Error: ${err ?? "Unknown"}'),
));

// One-line async loading (v2.0.0+): idle → loading → loaded / empty / error
// • the future never throws — errors are captured into status
// • previous items are preserved on failure
// • overlapping calls: only the latest result is applied (stale responses dropped)
Future<void> fetchItems() => items.load(() => api.getItems());

// Paging: appends WITHOUT flashing `loading`, sets hasMore = page.isNotEmpty
Future<void> nextPage() => items.loadMore(() => api.getItems(page: ++_page));

// Manual control is still available when you need it
void fetchItemsManually() async {
  try {
    items.setLoading();
    items.assignAll(await api.getItems()); // → loaded (or empty)
  } catch (e) {
    items.setError(e.toString());          // keeps items, status → error
  }
}

// ==========================================
// 2. RxS (Status-Aware Nullable Value)
// ==========================================
final user = RxS<User?>(null);       // starts as idle

Obx(() => user.on(
  idle: () => const Text('Idle...'),
  loading: () => const CircularProgressIndicator(),
  loaded: (data) => Text('Hello, ${data?.name ?? "Guest"}'),
  error: (err) => Text('Error: ${err ?? "Unknown"}'),
));

// One-line async loading with a friendly error mapper (v2.0.0+)
Future<void> fetchUser() => user.load(
  () => api.getUser(),
  errorMessage: (e) => 'Could not load profile',
);
