class SearchController extends GetxController {
  final searchQuery = ''.obs;
  final page = 0.obs;
  final sort = 'name'.obs;
  final ticks = 0.obs;
  final _workers = <Worker>[];

  @override
  void onInit() {
    super.onInit();
    _workers.addAll([
      // Fires only after 500ms of typing inactivity
      debounce(searchQuery, (q) => fetchApi(q),
          time: const Duration(milliseconds: 500)),

      // v2.0.0+: at most once per window (first value wins) — anti double-tap
      interval(page, (p) => loadPage(p), time: const Duration(seconds: 1)),

      // v2.0.0+: any of several observables changed
      everAll([page, sort], (_) => refreshList()),

      // Every change; onError also receives failures from updateSequential / load()
      ever(searchQuery, (q) => log(q), onError: (e) => report(e)),
    ]);

    // v2.0.0+: drive an Rx from a Stream — cancelled automatically on close()
    ticks.bindStream(Stream.periodic(const Duration(seconds: 1), (i) => i));
  }

  @override
  void onClose() {
    for (final w in _workers) {
      w.dispose(); // explicit disposal prevents leaks
    }
    super.onClose();
  }
}
